cmake_minimum_required(VERSION 3.21)
project(slopmeter LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2 Widgets DBus)
find_package(LayerShellQt REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
qt_add_executable(slopmeter main.cpp)
file(GLOB CLASS_ICONS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS "icons/*.svg")
file(GLOB GAME_ICONS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS "class-icons/*.png" "skill-icons/*.webp")
qt_add_resources(slopmeter qml PREFIX "/" FILES Main.qml FightDetails.qml Settings.qml preview.json game-screen.js ${CLASS_ICONS} ${GAME_ICONS})
target_link_libraries(slopmeter PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::DBus LayerShellQt::Interface PkgConfig::WAYLAND)

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" SLOPMETER_DEFAULT_VERSION LIMIT_COUNT 1)
set(SLOPMETER_VERSION "${SLOPMETER_DEFAULT_VERSION}" CACHE STRING "SlopMeter version")
target_compile_definitions(slopmeter PRIVATE SLOPMETER_VERSION="${SLOPMETER_VERSION}")
install(TARGETS slopmeter RUNTIME DESTINATION bin)
# Debian 13 ships LayerShellQt 6.3; newer versions expose explicit size/screen setters.
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES Qt6::Gui LayerShellQt::Interface)
check_cxx_source_compiles("#include <LayerShellQt/Window>\nint main() { LayerShellQt::Window::get(nullptr)->setDesiredSize(QSize(0,0)); }" SLOPMETER_LAYER_DESIRED_SIZE)
check_cxx_source_compiles("#include <LayerShellQt/Window>\nint main() { LayerShellQt::Window::get(nullptr)->setScreen(nullptr); }" SLOPMETER_LAYER_SCREEN)
unset(CMAKE_REQUIRED_LIBRARIES)
if(SLOPMETER_LAYER_DESIRED_SIZE)
    target_compile_definitions(slopmeter PRIVATE SLOPMETER_LAYER_DESIRED_SIZE)
endif()
if(SLOPMETER_LAYER_SCREEN)
    target_compile_definitions(slopmeter PRIVATE SLOPMETER_LAYER_SCREEN)
endif()
